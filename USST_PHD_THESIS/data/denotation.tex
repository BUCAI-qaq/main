\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}ll}
	\toprule
	符号 & 说明 \\
	\midrule
	
	$\mathbf{T}_{g}^{w}$ & 从坐标系 $g$ 到坐标系 $w$ 的齐次变换矩阵 \\
	$(\cdot)^T$ & 矩阵或向量的转置 \\
	$\frac{\partial}{\partial x}$ & 对变量 $x$ 的偏导算子 \\
	$\nabla$ & 梯度算子 \\
	$\log(\cdot)$ & 对数运算 \\
	$\exp(\cdot)$ & 指数运算 \\
	$\sum$ & 求和运算 \\
	
	$[\cdot]_\times$ & 向量的反对称矩阵（叉乘矩阵） \\
	$\times$ & 向量叉乘运算 \\
	$\odot$ & 逐元素乘积 \\
	$\ominus$ & 位姿误差运算（如 SE(3) 中的差分） \\
	
	$\|\cdot\|$ & 向量范数 \\
	$\|\cdot\|_2$ & 二范数 \\
	$\|\cdot\|_W$ & 加权范数 \\
	$\|\cdot\|^2$ & 二范数平方 \\
	
	$\mathrm{sgn}(\cdot)$ & 符号函数 \\
	
	$\mathcal{L}$ & 损失函数 \\
	$\mathbb{E}[\cdot]$ & 期望算子 \\
	
	$\mathbf{z}$ & 潜变量（低维动作表示） \\
	$\mathcal{E}(\cdot)$ & 编码器（Encoder） \\
	$\mathcal{D}(\cdot)$ & 解码器（Decoder） \\
	
	$\mathrm{s.t.}$ & 约束条件（subject to） \\
	
	\bottomrule
\end{tabular*}
